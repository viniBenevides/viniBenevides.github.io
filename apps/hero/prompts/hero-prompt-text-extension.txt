    <script>
        const settings = {
            darkAppearance: false,
            centreText: false,
            minimumHeight: true,
            showSubheading: true,
            h1Serif: false,
            h2Serif: false,
            h1Spacing: -0.05,
            h2Spacing: -0.025
        };

        const darkControl = document.querySelector("#dark-appearance");
        const centreControl = document.querySelector("#centre-text");
        const heightControl = document.querySelector("#minimum-height");
        const subheadingControl = document.querySelector("#show-subheading");

        const h1SerifControl = document.querySelector("#h1-serif");
        const h2SerifControl = document.querySelector("#h2-serif");
        const h1SpacingControl = document.querySelector("#h1-spacing");
        const h2SpacingControl = document.querySelector("#h2-spacing");

        const h1SpacingValue = document.querySelector("#h1-spacing-value");
        const h2SpacingValue = document.querySelector("#h2-spacing-value");
        const h2Effects = document.querySelector("#h2-effects");

        const hero = document.querySelector("#hero");
        const heading = document.querySelector("#hero-heading");
        const subheading = document.querySelector("#hero-subheading");
        const heroStyles = document.querySelector("#hero-styles");
        const heroSpacing = document.querySelector("#hero-spacing");

        const copyButton = document.querySelector("#copy-button");
        const copyStatus = document.querySelector("#copy-status");
        const manualCopy = document.querySelector("#manual-copy");
        const htmlOutput = document.querySelector("#html-output");

        function updateSettings() {
            settings.darkAppearance = darkControl.checked;
            settings.centreText = centreControl.checked;
            settings.minimumHeight = heightControl.checked;
            settings.showSubheading = subheadingControl.checked;

            settings.h1Serif = h1SerifControl.checked;
            settings.h2Serif = h2SerifControl.checked;

            // Range inputs provide strings; convert them to numbers.
            settings.h1Spacing = Number(h1SpacingControl.value);
            settings.h2Spacing = Number(h2SpacingControl.value);

            updateHero();
            refreshOutput();
        }

        function updateHero() {
            // Keep the existing editable elements and their text.
            hero.classList.toggle("hero-light", !settings.darkAppearance);
            hero.classList.toggle("hero-dark", settings.darkAppearance);
            hero.classList.toggle("hero-centre", settings.centreText);
            hero.classList.toggle("hero-min-height", settings.minimumHeight);

            subheading.hidden = !settings.showSubheading;
            h2Effects.disabled = !settings.showSubheading;

            heading.classList.toggle("heading-sans", !settings.h1Serif);
            heading.classList.toggle("heading-serif", settings.h1Serif);

            subheading.classList.toggle("heading-sans", !settings.h2Serif);
            subheading.classList.toggle("heading-serif", settings.h2Serif);

            updateLetterSpacing();
        }

        function updateLetterSpacing() {
            const h1Value = settings.h1Spacing + "em";
            const h2Value = settings.h2Spacing + "em";

            h1SpacingValue.value = h1Value;
            h2SpacingValue.value = h2Value;

            h1SpacingControl.setAttribute("aria-valuetext", h1Value);
            h2SpacingControl.setAttribute("aria-valuetext", h2Value);

            // Update CSS rules in the head, never inline styles.
            heroSpacing.textContent = `
        .hero h1 {
            letter-spacing: ${h1Value};
        }

        .hero h2 {
            letter-spacing: ${h2Value};
        }
    `;
        }

        function escapeHTML(text) {
            return text
                .replaceAll("&", "&amp;")
                .replaceAll("<", "&lt;")
                .replaceAll(">", "&gt;")
                .replaceAll('"', "&quot;")
                .replaceAll("'", "&#39;");
        }

        function generateHTML() {
            const headingText = escapeHTML(heading.innerText);
            const description = document.querySelector(
                'meta[name="description"]'
            ).content;

            let subheadingHTML = "";

            if (settings.showSubheading) {
                const subheadingText = escapeHTML(subheading.innerText);

                subheadingHTML =
                    `\n            <h2 class="${subheading.className}">${subheadingText}</h2>`;
            }

            // Export both shared stylesheets and the selected font classes.
            return `<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <meta name="description" content="${escapeHTML(description)}">
    <title>My Hero Page</title>

    <script src="https://cdn.jsdelivr.net/npm/@tailwindcss/browser@4"><\/script>

    <style type="text/tailwindcss">${heroStyles.textContent}
    </style>

    <style>${heroSpacing.textContent}
    </style>
</head>

<body>
    <header class="${hero.className}">
        <div class="hero-content">
            <h1 class="${heading.className}">${headingText}</h1>${subheadingHTML}
        </div>
    </header>
</body>
</html>`;
        }

        function refreshOutput() {
            copyStatus.textContent = "";

            if (!manualCopy.hidden) {
                htmlOutput.value = generateHTML();
            }
        }

        async function copyHTML() {
            const html = generateHTML();

            try {
                if (!navigator.clipboard?.writeText) {
                    throw new Error("Clipboard API unavailable");
                }

                await navigator.clipboard.writeText(html);

                manualCopy.hidden = true;
                copyStatus.textContent =
                    "HTML copied—paste it into VS Code.";
            } catch (error) {
                manualCopy.hidden = false;
                htmlOutput.value = generateHTML();

                manualCopy.scrollIntoView({
                    behavior: "smooth",
                    block: "center"
                });

                htmlOutput.focus({ preventScroll: true });
                htmlOutput.select();

                copyStatus.textContent =
                    "Copy the selected HTML manually.";
            }
        }

        function insertPlainText(element, text) {
            const selection = window.getSelection();

            if (!selection || selection.rangeCount === 0) {
                return;
            }

            const range = selection.getRangeAt(0);

            if (!element.contains(range.commonAncestorContainer)) {
                return;
            }

            range.deleteContents();

            const textNode = document.createTextNode(text);
            range.insertNode(textNode);
            range.setStartAfter(textNode);
            range.collapse(true);

            selection.removeAllRanges();
            selection.addRange(range);

            refreshOutput();
        }

        function handlePaste(event) {
            event.preventDefault();

            const text = event.clipboardData.getData("text/plain");
            insertPlainText(event.currentTarget, text);
        }

        function preventDrop(event) {
            event.preventDefault();
        }

        darkControl.addEventListener("change", updateSettings);
        centreControl.addEventListener("change", updateSettings);
        heightControl.addEventListener("change", updateSettings);
        subheadingControl.addEventListener("change", updateSettings);

        h1SerifControl.addEventListener("change", updateSettings);
        h2SerifControl.addEventListener("change", updateSettings);

        // The input event updates spacing continuously while dragging.
        h1SpacingControl.addEventListener("input", updateSettings);
        h2SpacingControl.addEventListener("input", updateSettings);

        heading.addEventListener("input", refreshOutput);
        subheading.addEventListener("input", refreshOutput);

        heading.addEventListener("paste", handlePaste);
        subheading.addEventListener("paste", handlePaste);

        heading.addEventListener("drop", preventDrop);
        subheading.addEventListener("drop", preventDrop);

        copyButton.addEventListener("click", copyHTML);

        updateSettings();
    </script>
